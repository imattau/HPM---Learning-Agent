# Structural Creativity Benchmark (SCB) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an SCB that tests HPM v5's ability to generate novel, syntactically valid Python skeletons by extending learned U_* sequences via PatternSequence.simulate().

**Architecture:** Stdlib functions are extracted to a corpus, tokenised through NL2CodeBridgeAdapter to produce U_* sequences, and fed to PatternEngine for training. A new UCodeRenderer postprocessor maps U_* sequences to Python skeletons. The benchmark holds out 20% of functions, seeds generation from their first 1-2 U_* tokens, and scores correctness/novelty/recombination.

**Tech Stack:** Python 3.12, spaCy en_core_web_sm, ast module, uv, pytest

"""Extract short Python functions from stdlib for SCB corpus."""
from __future__ import annotations
import ast
import inspect
import textwrap
import os
import io
import pathlib


def _get_module_functions(module) -> list[str]:
    results = []
    try:
        src = inspect.getsource(module)
    except (OSError, TypeError):
        return results
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return results
    lines = src.splitlines()
    for node in ast.walk(tree):
        if not isinstance(node, ast.FunctionDef):
            continue
        if not (3 <= len(node.body) <= 20):
            continue
        start = node.lineno - 1
        end = node.end_lineno
        fn_src = textwrap.dedent("\n".join(lines[start:end]))
        try:
            ast.parse(fn_src)
            results.append(fn_src)
        except SyntaxError:
            continue
    return results


def extract_functions(max_functions: int = 200) -> list[str]:
    """Extract up to max_functions short functions from os, io, pathlib."""
    results: list[str] = []
    for module in [os, os.path, io, pathlib]:
        results.extend(_get_module_functions(module))
        if len(results) >= max_functions:
            break
    return results[:max_functions]
```

"""UCodeRenderer — maps U_* sequences to Python function skeletons."""
from __future__ import annotations
from dataclasses import dataclass

_TEMPLATES: dict[str, str] = {
    "U_IF":     "    if condition:\n        pass",
    "U_WHILE":  "    while condition:\n        pass",
    "U_FOR":    "    for item in iterable:\n        pass",
    "U_TRY":    "    try:\n        pass\n    except Exception:\n        pass",
    "U_ASSIGN": "    result = value",
    "U_RETURN": "    return result",
    "U_CALL":   "    func()",
    "U_THROW":  "    raise Exception()",
}


@dataclass
class UCodeRenderer:
    """Renders a sequence of U_* tokens to a valid Python function skeleton."""

    def render(self, tokens: list[str]) -> str:
        lines = ["def generated_fn():"]
        for token in tokens:
            snippet = _TEMPLATES.get(token, "    pass")
            lines.append(snippet)
        if not tokens:
            lines.append("    pass")
        return "\n".join(lines)
```

```python
"""Structural Creativity Benchmark (SCB) for HPM v5."""
from __future__ import annotations
import ast
import random
from collections import defaultdict

from hpm_ai_v5.adapter.nlp import NLPTokenizer, NL2CodeBridgeAdapter
from hpm_ai_v5.adapter.validation_only import ValidationOnlyAdapter
from hpm_ai_v5.core import PatternEngine, PatternManager, PatternStore
from hpm_ai_v5.core.config import CoreConfig
from hpm_ai_v5.experiments.corpus.extract_stdlib import extract_functions
from hpm_ai_v5.pipeline import HPMPipeline
from hpm_ai_v5.polygraphs.code import CodePolygraphGenerator
from hpm_ai_v5.postprocessors.code import UCodeRenderer


U_TOKENS = {"U_IF", "U_WHILE", "U_FOR", "U_TRY", "U_ASSIGN", "U_RETURN", "U_CALL", "U_THROW"}


def _extract_u_sequence(text: str, bridge: NL2CodeBridgeAdapter, tokenizer: NLPTokenizer) -> list[str]:
    """Run NLPTokenizer + NL2CodeBridgeAdapter on raw text, return list of U_* tokens."""
    from hpm_ai_v5.adapter.packet import AdapterPacket
    packet = AdapterPacket(raw=text, context={}, states=[])
    packet = tokenizer.run(packet)
    packet = bridge.run(packet)
    return [t for t in packet.context.get("bridge_tokens", []) if t in U_TOKENS]


class SCBBenchmark:
    def __init__(self):
        self.config = CoreConfig(
            max_patterns=512,
            max_sequences=256,
            history_limit=50,
            near_threshold=0.4,
        )
        self.engine = PatternEngine(config=self.config)
        self.manager = PatternManager(promotion_threshold=0.01)
        self.renderer = UCodeRenderer()
        self.tokenizer = NLPTokenizer()
        self.bridge = NL2CodeBridgeAdapter()

        self.pipeline = HPMPipeline(
            preprocessor=NLPTokenizer(),
            engine=self.engine,
            postprocessor=ValidationOnlyAdapter(),
            polygraph_generator=CodePolygraphGenerator(),
            polygraph_confidence_skip=1.1,
        )
        self.pipeline.register_preprocessor(NL2CodeBridgeAdapter())
        # training_sequences[pattern_name] = list of training U_* sequences
        self.training_sequences: list[list[str]] = []
        self.pattern_to_training_idx: dict[str, int] = {}

    def _reset(self):
        self.engine.current_state = None
        self.engine.history = []
        self.pipeline.view_engines.clear()
        self.engine.store = PatternStore(config=self.config)
        self.training_sequences.clear()
        self.pattern_to_training_idx.clear()
        for adapter in self.pipeline.preprocessing_pipeline.adapters.values():
            if hasattr(adapter, "reset"):
                adapter.reset()

    def train(self, functions: list[str]):
        self._reset()
        self.manager.start_episode(self.engine)
        for i, fn_src in enumerate(functions):
            self.engine.current_state = None
            self.engine.history = []
            self.pipeline.step(fn_src)
            u_seq = _extract_u_sequence(fn_src, self.bridge, self.tokenizer)
            if u_seq:
                idx = len(self.training_sequences)
                self.training_sequences.append(u_seq)
                if self.engine.last_match and self.engine.last_match.pattern:
                    self.pattern_to_training_idx[self.engine.last_match.pattern.name] = idx
        self.manager.end_episode(self.engine)

    def generate(self, seed_tokens: list[str]) -> list[str]:
        """Extend seed U_* sequence using PatternSequence.simulate()."""
        if not seed_tokens:
            return []
        # Step the engine with the seed to set state
        self.engine.current_state = None
        self.engine.history = []
        for tok in seed_tokens:
            self.pipeline.step(tok)
        # Select best sequence and simulate
        seq = self.engine.select_sequence()
        if seq is None:
            return seed_tokens[:]
        path = seq.simulate(
            self.engine.current_state,
            horizon=6,
            resolver=self.engine.store.get,
            start_offset=0,
        )
        # Recover U_* tokens from simulated states
        generated = list(seed_tokens)
        for state in path:
            if state and state.context:
                btokens = state.context.get("bridge_tokens", [])
                generated.extend(t for t in btokens if t in U_TOKENS)
        return generated[:8]  # cap length

    def _is_novel(self, seq: list[str]) -> bool:
        return seq not in self.training_sequences

    def _recombination_count(self, seq: list[str]) -> int:
        """Count distinct training patterns whose transitions appear in seq."""
        if len(seq) < 2:
            return 0
        bigrams = set(zip(seq, seq[1:]))
        sources: set[int] = set()
        for train_seq in self.training_sequences:
            train_bigrams = set(zip(train_seq, train_seq[1:]))
            if bigrams & train_bigrams:
                sources.add(id(train_seq))
        return len(sources)

    def evaluate(self, held_out: list[str]) -> dict:
        correct = novel = 0
        recombination_total = 0
        n = len(held_out)

        for fn_src in held_out:
            u_seq = _extract_u_sequence(fn_src, self.bridge, self.tokenizer)
            if not u_seq:
                n -= 1
                continue
            seed = u_seq[:2]
            generated = self.generate(seed)
            rendered = self.renderer.render(generated)
            try:
                ast.parse(rendered)
                correct += 1
            except SyntaxError:
                pass
            if self._is_novel(generated):
                novel += 1
            recombination_total += self._recombination_count(generated)

        if n == 0:
            return {"n": 0}

        # Baseline: random sequences of average length
        avg_len = int(sum(len(_extract_u_sequence(f, self.bridge, self.tokenizer)) for f in held_out) / max(n, 1))
        u_list = list(U_TOKENS)
        baseline_correct = 0
        for _ in range(n):
            rand_seq = [random.choice(u_list) for _ in range(max(avg_len, 2))]
            try:
                ast.parse(self.renderer.render(rand_seq))
                baseline_correct += 1
            except SyntaxError:
                pass

        return {
            "n": n,
            "correctness": correct / n,
            "novelty": novel / n,
            "avg_recombination": recombination_total / n,
            "baseline_correctness": baseline_correct / n,
        }

    def run(self):
        print("Loading corpus...")
        all_fns = extract_functions(max_functions=200)
        random.shuffle(all_fns)
        split = int(len(all_fns) * 0.8)
        train_fns, held_out = all_fns[:split], all_fns[split:]
        print(f"  Train: {len(train_fns)}, Held-out: {len(held_out)}")

        print("Training...")
        self.train(train_fns)
        print(f"  Patterns learned: {len(self.engine.store.patterns)}")
        print(f"  Sequences learned: {len(list(self.engine.store_sequences() if hasattr(self.engine, 'store_sequences') else []))}")

        print("Generating and scoring...")
        results = self.evaluate(held_out)

        print("\n" + "=" * 40)
        print("SCB RESULTS")
        print("=" * 40)
        print(f"N (held-out with U_* signal): {results['n']}")
        print(f"Correctness:    {results.get('correctness', 0):.2%}  (target >90%)")
        print(f"Novelty:        {results.get('novelty', 0):.2%}  (target >50%)")
        print(f"Recombination:  {results.get('avg_recombination', 0):.2f} sources  (target >=2 in 60%)")
        print(f"Baseline corr:  {results.get('baseline_correctness', 0):.2%}  (random floor)")
        print("=" * 40)
        return results


if __name__ == "__main__":
    SCBBenchmark().run()
```
